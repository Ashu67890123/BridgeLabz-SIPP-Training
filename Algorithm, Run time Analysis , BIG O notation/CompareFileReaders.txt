import java.io.BufferedReader;
import java.io.FileReader;
import java.io.InputStreamReader;
import java.io.FileInputStream;
import java.nio.charset.StandardCharsets;
import java.util.Scanner;

/**
 * Compares FileReader and InputStreamReader for reading large files.
 * Follows best practices: wrap with BufferedReader, specify charset, measure performance.
 * Adapts cab service method: enhanceInvoice (displayComparison).
 */
public class CompareFileReaders {
    // Read file with FileReader: O(n)
    public long readFileReader(String fileName) {
        long startTime = System.nanoTime();
        try (BufferedReader br = new BufferedReader(new FileReader(fileName))) {
            while (br.readLine() != null) {
                // Read entire file
            }
        } catch (Exception e) {
            System.err.println("Error reading file: " + e.getMessage());
        }
        return System.nanoTime() - startTime;
    }

    // Read file with InputStreamReader: O(n)
    public long readInputStreamReader(String fileName) {
        long startTime = System.nanoTime();
        try (BufferedReader br = new BufferedReader(
                new InputStreamReader(new FileInputStream(fileName), StandardCharsets.UTF_8))) {
            while (br.readLine() != null) {
                // Read entire file
            }
        } catch (Exception e) {
            System.err.println("Error reading file: " + e.getMessage());
        }
        return System.nanoTime() - startTime;
    }

    // Run readers and measure time: O(n)
    public String runReaders(String fileName) {
        long fileReaderTime = readFileReader(fileName);
        long inputStreamReaderTime = readInputStreamReader(fileName);
        return enhanceInvoice(fileReaderTime, inputStreamReaderTime);
    }

    // Display comparison: O(1)
    public String enhanceInvoice(long fileReaderTime, long inputStreamReaderTime) {
        StringBuilder sb = new StringBuilder("File Reading Performance:\n");
        sb.append("----------------\n");
        sb.append(String.format("FileReader Time: %d ns\n", fileReaderTime));
        sb.append(String.format("InputStreamReader Time: %d ns\n", inputStreamReaderTime));
        sb.append("----------------\n");
        return sb.toString();
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        CompareFileReaders reader = new CompareFileReaders();

        System.out.print("Enter file path: ");
        String fileName = scanner.nextLine();

        System.out.println(reader.runReaders(fileName));

        scanner.close();
    }
}
