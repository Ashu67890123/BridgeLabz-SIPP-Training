import java.util.Arrays;
import java.util.Random;
import java.util.Scanner;

/**
 * Compares Bubble Sort (O(N²)), Merge Sort (O(N log N)), and Quick Sort (O(N log N)).
 * Follows best practices: use efficient sorting (Merge/Quick), measure performance.
 * Adapts cab service method: enhanceInvoice (displayComparison).
 */
public class CompareSortingAlgorithms {
    // Bubble Sort: O(N²)
    public void bubbleSort(int[] arr) {
        int n = arr.length;
        for (int i = 0; i < n - 1; i++) {
            boolean swapped = false;
            for (int j = 0; j < n - i - 1; j++) {
                if (arr[j] > arr[j + 1]) {
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                    swapped = true;
                }
            }
            if (!swapped) break;
        }
    }

    // Merge Sort: O(N log N)
    public void mergeSort(int[] arr, int left, int right) {
        if (left < right) {
            int mid = left + (right - left) / 2;
            mergeSort(arr, left, mid);
            mergeSort(arr, mid + 1, right);
            merge(arr, left, mid, right);
        }
    }

    private void merge(int[] arr, int left, int mid, int right) {
        int n1 = mid - left + 1, n2 = right - mid;
        int[] L = new int[n1], R = new int[n2];
        for (int i = 0; i < n1; i++) L[i] = arr[left + i];
        for (int i = 0; i < n2; i++) R[i] = arr[mid + 1 + i];
        int i = 0, j = 0, k = left;
        while (i < n1 && j < n2) arr[k++] = L[i] <= R[j] ? L[i++] : R[j++];
        while (i < n1) arr[k++] = L[i++];
        while (j < n2) arr[k++] = R[j++];
    }

    // Quick Sort: O(N log N) average
    public void quickSort(int[] arr, int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }

    private int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;
        for (int j = low; j < high; j++) {
            if (arr[j] <= pivot) {
                i++;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;
        return i + 1;
    }

    // Run sorts and measure time: O(N²) or O(N log N)
    public String runSorts(int[] arr) {
        long bubbleTime, mergeTime, quickTime;

        // Bubble Sort
        int[] bubbleArr = arr.clone();
        long startTime = System.nanoTime();
        bubbleSort(bubbleArr);
        bubbleTime = System.nanoTime() - startTime;

        // Merge Sort
        int[] mergeArr = arr.clone();
        startTime = System.nanoTime();
        mergeSort(mergeArr, 0, mergeArr.length - 1);
        mergeTime = System.nanoTime() - startTime;

        // Quick Sort
        int[] quickArr = arr.clone();
        startTime = System.nanoTime();
        quickSort(quickArr, 0, quickArr.length - 1);
        quickTime = System.nanoTime() - startTime;

        return enhanceInvoice(arr.length, bubbleTime, mergeTime, quickTime);
    }

    // Display comparison: O(1)
    public String enhanceInvoice(int size, long bubbleTime, long mergeTime, long quickTime) {
        StringBuilder sb = new StringBuilder("Sorting Performance Comparison:\n");
        sb.append("----------------\n");
        sb.append(String.format("Dataset Size: %d\n", size));
        sb.append(String.format("Bubble Sort Time: %d ns\n", bubbleTime));
        sb.append(String.format("Merge Sort Time: %d ns\n", mergeTime));
        sb.append(String.format("Quick Sort Time: %d ns\n", quickTime));
        sb.append("----------------\n");
        return sb.toString();
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        CompareSortingAlgorithms sorter = new CompareSortingAlgorithms();
        Random random = new Random();

        System.out.print("Enter dataset size: ");
        int n = scanner.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = random.nextInt(10000); // Random integers
        }

        System.out.println(sorter.runSorts(arr));

        scanner.close();
    }
}
